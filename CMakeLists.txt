# Optional views and host applications. Only applications require Telemetry SDK. MIT.
cmake_minimum_required(VERSION 3.21)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" TELEMETRY_UX_VERSION LIMIT_COUNT 1)
project(TelemetryUX VERSION ${TELEMETRY_UX_VERSION} LANGUAGES CXX)
option(TELEMETRY_UX_BUILD_INSPECTOR "Build the optional Universal Inspector example" OFF)
option(TELEMETRY_UX_BUILD_CONFIGURATOR "Build and install the reusable Qt/QML view" ON)
option(TELEMETRY_UX_BUILD_DYNAMIC_CONFIGURATOR "Build the native runtime Qt provider (requires SDK)" OFF)
option(TELEMETRY_UX_BUILD_QT_CONNECTION "Build automatic SDK Qt Session presentation binding" OFF)
if(TELEMETRY_UX_BUILD_QT_CONNECTION)
    set(TELEMETRY_UX_BUILD_DYNAMIC_CONFIGURATOR ON)
endif()
if(TELEMETRY_UX_BUILD_INSPECTOR)
    set(TELEMETRY_UX_BUILD_DYNAMIC_CONFIGURATOR ON)
endif()
if(TELEMETRY_UX_BUILD_CONFIGURATOR OR TELEMETRY_UX_BUILD_INSPECTOR OR TELEMETRY_UX_BUILD_DYNAMIC_CONFIGURATOR)
    add_subdirectory(lib/qml)
endif()
if(TELEMETRY_UX_BUILD_INSPECTOR)
    add_subdirectory(examples/inspector)
endif()
