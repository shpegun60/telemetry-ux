# Observe the installed public find_package contract without Qt. MIT.
cmake_minimum_required(VERSION 3.21)
project(TelemetryUXConfigControls LANGUAGES CXX)
if(CASE STREQUAL "optional")
    find_package(TelemetryUX 0.1.0 CONFIG REQUIRED OPTIONAL_COMPONENTS configurator)
    if(TelemetryUX_configurator_FOUND)
        message(FATAL_ERROR "Unavailable optional Qt component was reported found")
    endif()
elseif(CASE STREQUAL "optional-unknown")
    find_package(TelemetryUX 0.1.0 CONFIG REQUIRED OPTIONAL_COMPONENTS unknown)
    if(TelemetryUX_unknown_FOUND)
        message(FATAL_ERROR "Unknown optional component was reported found")
    endif()
elseif(CASE STREQUAL "unknown")
    find_package(TelemetryUX 0.1.0 CONFIG REQUIRED COMPONENTS unknown)
elseif(CASE STREQUAL "next-minor")
    # 0.x: the next minor may break, so it is refused.
    find_package(TelemetryUX 0.2 CONFIG REQUIRED OPTIONAL_COMPONENTS configurator)
elseif(CASE STREQUAL "default")
    find_package(TelemetryUX 0.1.0 CONFIG REQUIRED)
else()
    find_package(TelemetryUX 0.1.0 CONFIG REQUIRED COMPONENTS configurator)
endif()
