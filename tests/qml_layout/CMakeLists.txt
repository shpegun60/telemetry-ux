# Geometry regression over the real disconnected Session presentation. MIT.
cmake_minimum_required(VERSION 3.25)
project(telemetry_qml_layout LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Test)
find_package(Telemetry 0.1.0 CONFIG REQUIRED COMPONENTS qt_connection workbench core encoded_server encoded_backend)
set(TELEMETRY_UX_BUILD_DYNAMIC_CONFIGURATOR ON)
set(TELEMETRY_UX_BUILD_QT_CONNECTION ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../lib/qml" ux-qml)
qt_add_executable(telemetry_qml_layout_check main.cpp)
qt_add_resources(telemetry_qml_layout_check "layout-entry" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES Main.qml Standalone.qml Narrow.qml Search.qml EndpointStates.qml)
target_compile_features(telemetry_qml_layout_check PRIVATE cxx_std_20)
target_link_libraries(telemetry_qml_layout_check PRIVATE TelemetryUX::qt_connection Qt6::Test)
qt_add_executable(telemetry_qml_search_providers SearchProviders.cpp)
qt_add_resources(telemetry_qml_search_providers "search-entry" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES SearchNative.qml)
target_compile_features(telemetry_qml_search_providers PRIVATE cxx_std_20)
target_compile_definitions(telemetry_qml_search_providers PRIVATE QT_NO_KEYWORDS)
target_link_libraries(telemetry_qml_search_providers PRIVATE TelemetryUX::dynamic_configurator
    TelemetryUX::static_configurator Telemetry::core Telemetry::encoded_server Telemetry::encoded_backend Qt6::Test)
qt_add_executable(telemetry_qml_bulk_selection Bulk.cpp)
set_target_properties(telemetry_qml_bulk_selection PROPERTIES AUTOMOC ON)
qt_add_resources(telemetry_qml_bulk_selection "bulk-entry" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES Bulk.qml)
target_compile_features(telemetry_qml_bulk_selection PRIVATE cxx_std_20)
target_link_libraries(telemetry_qml_bulk_selection PRIVATE TelemetryUX::qt_connection Qt6::Test)
qt_add_executable(telemetry_qml_trend Trend.cpp)
set_target_properties(telemetry_qml_trend PROPERTIES AUTOMOC ON)
qt_add_resources(telemetry_qml_trend "trend-entry" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES Trend.qml)
target_compile_features(telemetry_qml_trend PRIVATE cxx_std_20)
target_link_libraries(telemetry_qml_trend PRIVATE TelemetryUX::qt_connection Qt6::Test)
qt_add_executable(telemetry_qml_tree_actions TreeActions.cpp)
set_target_properties(telemetry_qml_tree_actions PROPERTIES AUTOMOC ON)
qt_add_resources(telemetry_qml_tree_actions "tree-actions-entry" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES TreeActions.qml)
target_compile_features(telemetry_qml_tree_actions PRIVATE cxx_std_20)
target_link_libraries(telemetry_qml_tree_actions PRIVATE TelemetryUX::qt_connection Qt6::Test)
qt_add_executable(telemetry_qml_theme ThemeTypes.cpp)
qt_add_resources(telemetry_qml_theme "theme-entry" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}" FILES ThemeTypes.qml)
target_compile_features(telemetry_qml_theme PRIVATE cxx_std_20)
target_link_libraries(telemetry_qml_theme PRIVATE TelemetryUX::configurator Qt6::Quick)
