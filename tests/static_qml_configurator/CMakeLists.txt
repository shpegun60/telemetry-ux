cmake_minimum_required(VERSION 3.25)
project(telemetry_static_qml_provider_checks LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Qml Quick Test)
find_package(Telemetry 0.1.0 CONFIG REQUIRED COMPONENTS core encoded_server encoded_backend)
find_package(TelemetryUX 0.1.0 CONFIG REQUIRED COMPONENTS static_configurator)
qt_standard_project_setup(REQUIRES 6.12)
if(NOT GENERATED_SDK)
    message(FATAL_ERROR "Set GENERATED_SDK to the opt-in generated SDK root")
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../examples/codegen_qml/fixture" fixture)
add_subdirectory("${GENERATED_SDK}/qml" generated)
qt_add_executable(telemetry_static_qml_provider main.cpp)
target_compile_features(telemetry_static_qml_provider PRIVATE cxx_std_20)
target_compile_definitions(telemetry_static_qml_provider PRIVATE QT_NO_KEYWORDS)
target_link_libraries(telemetry_static_qml_provider PRIVATE telemetry_qml_demo_fixture
    telemetry_qml_device_api TelemetryUX::static_configurator Qt6::Test
    Telemetry::encoded_server Telemetry::encoded_backend)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Contracts.cpp")
    qt_add_executable(telemetry_static_qml_contracts Contracts.cpp)
    target_compile_features(telemetry_static_qml_contracts PRIVATE cxx_std_20)
    target_link_libraries(telemetry_static_qml_contracts PRIVATE TelemetryUX::static_configurator Qt6::Test)
endif()
