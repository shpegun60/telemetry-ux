cmake_minimum_required(VERSION 3.25)
project(telemetry_static_qml_generated_checks LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core Qml Test)
find_package(Telemetry 0.1.0 CONFIG REQUIRED COMPONENTS core encoded_server encoded_backend)
find_package(TelemetryUX 0.1.0 CONFIG REQUIRED COMPONENTS static_configurator)
qt_standard_project_setup(REQUIRES 6.12)
if(NOT GENERATED_SDK)
    message(FATAL_ERROR "Set GENERATED_SDK to the opt-in generated SDK root")
endif()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../examples/codegen_qml/fixture" fixture)
add_subdirectory("${GENERATED_SDK}/qml" generated)
qt_add_executable(telemetry_static_qml_generated main.cpp)
target_compile_features(telemetry_static_qml_generated PRIVATE cxx_std_20)
target_compile_definitions(telemetry_static_qml_generated PRIVATE QT_NO_KEYWORDS)
target_link_libraries(telemetry_static_qml_generated PRIVATE telemetry_qml_demo_fixture
    telemetry_qml_device_api Qt6::Test Telemetry::encoded_server Telemetry::encoded_backend)
if(GENERATED_NAMING_SDK)
    add_subdirectory("${GENERATED_NAMING_SDK}/qml" naming-generated)
    qt_add_executable(telemetry_static_qml_naming Naming.cpp)
    target_compile_features(telemetry_static_qml_naming PRIVATE cxx_std_20)
    target_link_libraries(telemetry_static_qml_naming PRIVATE telemetry_qml_naming_api)
endif()
