# Installed Qt-owned scheduling checks; no transport/controller implementation. MIT.
cmake_minimum_required(VERSION 3.21)
project(telemetry_dynamic_watchlist_checks LANGUAGES CXX)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Test)
find_package(TelemetryUX 0.1.0 CONFIG REQUIRED COMPONENTS dynamic_configurator)
qt_standard_project_setup(REQUIRES 6.4)
qt_add_executable(telemetry_dynamic_watchlist_check main.cpp)
target_compile_features(telemetry_dynamic_watchlist_check PRIVATE cxx_std_20)
target_compile_definitions(telemetry_dynamic_watchlist_check PRIVATE QT_NO_KEYWORDS)
target_link_libraries(telemetry_dynamic_watchlist_check PRIVATE TelemetryUX::dynamic_configurator Qt6::Test)
enable_testing()
add_test(NAME dynamic_watchlist COMMAND telemetry_dynamic_watchlist_check)
